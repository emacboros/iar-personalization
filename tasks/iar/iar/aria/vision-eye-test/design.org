import base64, json, urllib.request, time
img = base64.b64encode(open("/tmp/aria-eye/exterior_1_test.jpg","rb").read()).decode()
t0 = time.time()
req = urllib.request.Request("http://127.0.0.1:11434/api/chat",
    data=json.dumps({"model":"gemma4:31b","stream":False,
      "messages":[{"role":"user","content":"Describe this image. What do you see?","images":[img]}]}).encode(),
    headers={"Content-Type":"application/json"})
r = json.load(urllib.request.urlopen(req, timeout=900))
print("elapsed:", round(time.time()-t0,1), "s")
print(r["message"]["content"])
print("eval:", r.get("eval_count"), "tokens,", r.get("eval_duration",0)/1e9, "s")
EOF
